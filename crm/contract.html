function formatCurrency(amount) { return "$" + parseFloat(amount).toLocaleString('en-AU', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); } function generateInvoiceNumber(name, dateStr) { const initials = name.split(' ').map(n => n[0]).join('').toUpperCase().substring(0, 2); const datePart = dateStr.split('T')[0].replace(/-/g, ''); return `${initials}${datePart}`; } function populateInvoiceData(c) { const today = new Date(c.createdAt); const due = new Date(today); due.setDate(today.getDate() + 7); document.getElementById('outName').innerText = c.clientName; document.getElementById('outEmail').innerText = c.clientEmail; const formatDate = (d) => `${String(d.getDate()).padStart(2, '0')}/${String(d.getMonth() + 1).padStart(2, '0')}/${d.getFullYear()}`; document.getElementById('outDate').innerText = formatDate(today); document.getElementById('outDueDate').innerText = formatDate(due); document.getElementById('outInvoiceNum').innerText = generateInvoiceNumber(c.clientName, today.toISOString()); const tbody = document.getElementById('invoice-rows'); tbody.innerHTML = ''; let subtotal = 0; if (c.lineItems && Array.isArray(c.lineItems)) { c.lineItems.forEach(item => { // Because the backend modified the rates for inclusive GST, we just sum them strictly to 2 decimal places. const rowTotal = parseFloat((item.rate * item.qty).toFixed(2)); subtotal += rowTotal; const textColor = item.rate < 0 ? 'text-red-600' : 'text-gray-900'; const tr = document.createElement('tr'); tr.className = 'border-b'; tr.innerHTML = ` ${item.name || ''} ${item.desc || ''} ${item.qty} ${formatCurrency(item.rate)} ${formatCurrency(rowTotal)} `; tbody.appendChild(tr); }); } // Fallback for legacy database contracts let mode = c.gstMode; if (!mode) { mode = c.applyGst ? "exclusive" : "none"; } let gstAmount = 0; if (mode === "inclusive" || mode === "exclusive") { gstAmount = parseFloat((subtotal * 0.10).toFixed(2)); } const finalTotal = parseFloat((subtotal + gstAmount).toFixed(2)); document.getElementById('gstRow').style.display = (mode !== "none") ? 'flex' : 'none'; document.getElementById('outGst').innerText = formatCurrency(gstAmount); document.getElementById('outTotalTop').innerText = `AUD ${formatCurrency(finalTotal)}`; document.getElementById('outTotalBottom').innerText = formatCurrency(finalTotal); } window.onload = () => { initTimeline(); contentDisplay.classList.add('fade-enter-active'); loadContract(); };
